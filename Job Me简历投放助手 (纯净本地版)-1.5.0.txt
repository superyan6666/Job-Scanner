      const originalOpen = XMLHttpRequest.prototype.open;

      XMLHttpRequest.prototype.open = function(method, url) {
        this._jm_url = url;
        return originalOpen.apply(this, arguments);
      };
      XMLHttpRequest.prototype.send = function(data) {
        this.addEventListener('load', function() {
          if (this._jm_url) APIInterceptor.parseResponse(this._jm_url, this.responseText);
        });
        return originalSend.apply(this, arguments);
      };

      const originalFetch = window.fetch;
      window.fetch = async function(...args) {
        const response = await originalFetch.apply(this, args);
        try {
          const clone = response.clone();
          const url = typeof args[0] === 'string' ? args[0] : (args[0]?.url || '');
          clone.text().then(text => APIInterceptor.parseResponse(url, text)).catch(()=>{});
        } catch(e) {}
        return response;
      };
    },
    formatTime(val) {
      if (!val) return '';
      if (typeof val === 'number' || (typeof val === 'string' && /^\d{13}$/.test(val))) {
          const d = new Date(parseInt(val));
          return `${d.getMonth()+1}-${d.getDate()}`;
      }
      return val.toString().replace(/ 00:00:00$/, '').replace(/^\d{4}-/, '');
    },
    parseResponse(url, text) {
      try {
        if (!url.includes('joblist.json') && !url.includes('/search/joblist') && !url.includes('/api/job/search') && !url.includes('/recommend') && !url.includes('search-pc')) return;
        const data = JSON.parse(text);

        if (data?.zpData?.jobList) {
            data.zpData.jobList.forEach(job => {
                const time = job.activeTimeDesc || job.lastModifyTime || '';
                const formattedTime = this.formatTime(time);
                let timestamp = Date.now();
                if (time) {
                    if (typeof time === 'number') timestamp = time;
                    else if (typeof time === 'string' && /^\d{13}$/.test(time)) timestamp = parseInt(time);
                    else timestamp = new Date(time.toString().replace(/-/g, '/')).getTime() || Date.now();
                }
                APIDataCache.set(job.brandName, { updateTime: formattedTime, timestamp });
                APIDataCache.set(job.jobName + job.brandName, { updateTime: formattedTime, timestamp });
            });
        }
        if (data?.data?.list || data?.data?.results || data?.resultbody?.job?.items) {
            const list = data.data?.list || data.data?.results || data.resultbody?.job?.items;
            list.forEach(job => {
                const time = job.updateDate || job.publishTime || job.lastUpdate || job.updateDateTime || '';
                const cName = job.companyName || job.company?.name || job.fullCompanyName || '';
                const jName = job.name || job.jobName || job.jobTitle || '';
                const formattedTime = this.formatTime(time);
                let timestamp = Date.now();
                if (time) {
                    if (typeof time === 'number') timestamp = time;
                    else if (typeof time === 'string' && /^\d{13}$/.test(time)) timestamp = parseInt(time);
                    else timestamp = new Date(time.toString().replace(/-/g, '/')).getTime() || Date.now();
                }
                APIDataCache.set(cName, { updateTime: formattedTime, timestamp });
                if (jName) APIDataCache.set(jName + cName, { updateTime: formattedTime, timestamp });
            });
        }
      } catch(e) {}
    }

    getCleanText: (element) => {
        if (!element) return '';
        const clone = element.cloneNode(true);
        const styles = clone.querySelectorAll('style');
        const hiddenClasses = new Set();
        styles.forEach(style => {
            const regex = /\.([a-zA-Z0-9_-]+)[^{]*\{[^}]*(?:display:\s*none|visibility:\s*hidden|width:\s*0|font-size:\s*0)[^}]*\}/gi;
            let match;
            while ((match = regex.exec(style.textContent)) !== null) {
                if (match[1]) hiddenClasses.add(match[1]);
            }
            style.remove();
        });
        clone.querySelectorAll('span').forEach(span => {
            let isFake = false;
            span.classList.forEach(cls => { if (hiddenClasses.has(cls)) isFake = true; });
            if (isFake) span.remove();
        });
        clone.querySelectorAll('br').forEach(br => br.replaceWith('\n'));
        return clone.textContent ? clone.textContent.replace(/\n\s*\n/g, '\n\n').trim() : '';
    },
    parseQuickReplies: (str) => {
        const lines = (str || "").split('\n').map(l => l.trim()).filter(Boolean);
        const result = [];
        lines.forEach(line => {
            let catName = '默认';
            let repliesStr = line;
            if (line.includes(':')) {
                const parts = line.split(':');
                catName = parts[0];
                repliesStr = parts.slice(1).join(':');
            } else if (line.includes('：')) {
                const parts = line.split('：');
                catName = parts[0];
                repliesStr = parts.slice(1).join('：');
            }
            const replies = repliesStr.split(/[，,]/).map(i => i.trim()).filter(Boolean);
            result.push({ name: catName, items: replies });
        });
        if (result.length === 0) result.push({ name: '通用', items: [] });
        return result;

          }),

                resolve(data.choices[0].message.content);

                resolve(null);
              }
            } catch(e) { resolve(null); }
          },
          onerror: () => resolve(null)
        });
      });

  const WebDAVService = {
    getAuthHeader() {
      const { webdavUser, webdavPass } = state.settings;
      return "Basic " + btoa(unescape(encodeURIComponent(webdavUser + ":" + webdavPass)));
    },
    getFilePath() {

    },
    upload(data) {
      return new Promise((resolve, reject) => {
        if (!state.settings.webdavUrl || !state.settings.webdavUser) return reject(new Error("未配置 WebDAV 信息"));
        GM_xmlhttpRequest({
          method: "PUT",
          url: this.getFilePath(),
          headers: {
            "Authorization": this.getAuthHeader(),
            "Content-Type": "application/json;charset=UTF-8"
          },
          data: JSON.stringify(data, null, 2),
          onload: res => {
            if (res.status >= 200 && res.status < 300) resolve(true);
            else reject(new Error(`上传失败，HTTP 状态码: ${res.status}`));
          },
          onerror: () => reject(new Error("网络错误，无法连接到云盘"))
        });
      });
    },
    download() {
      return new Promise((resolve, reject) => {
        if (!state.settings.webdavUrl || !state.settings.webdavUser) return reject(new Error("未配置 WebDAV 信息"));
        GM_xmlhttpRequest({
          method: "GET",
          url: this.getFilePath(),
          headers: {
            "Authorization": this.getAuthHeader()
          },
          onload: res => {
            if (res.status === 404) return reject(new Error("云端尚未存在配置文件"));
            if (res.status >= 200 && res.status < 300) {
              try { resolve(JSON.parse(res.responseText)); } catch (e) { reject(new Error("云端配置文件解析失败")); }
            } else {
              reject(new Error(`下载失败，HTTP 状态码: ${res.status}`));
            }
          },
          onerror: () => reject(new Error("网络错误，无法连接到云盘"))
        });
      });
    }
  };

      const allElements = document.querySelectorAll("*");
      for (const el of allElements) {
        const text = (el.textContent || "").trim();
        if (text === "聊一聊" || text === "聊聊天") {

    _findCommunicateButtonByText(cardElement) {
      const PRIORITY = ["聊一聊", "开聊", "立即沟通", "沟通", "联系", "发消息"];
      const allElements = document.querySelectorAll("button, a, [role='button'], [class*='btn']");

      for (const keyword of PRIORITY) {
        for (const el of allElements) {
          const text = (el.textContent || "").trim();

            return el;
          }
        }
      }

    // 在卡片内查找沟通按钮
    _findButtonInCard(card) {
      if (!card) return null;

      const findBtn = (root) => {
        // 特定selector模式
        const SEL_IM = [
          ".xpath-open-im-btn",
          "[class*='im-btn']",
          "[class*='contact-btn']",
          "[class*='chat-btn']",
        ];
        for (const sel of SEL_IM) {
          const b = root.querySelector(sel);
          if (b && !b.disabled && this._isElementClickable(b)) return b;
        }

        // 文本匹配
        const PRIORITY = ["沟通", "聊一聊", "联系"];
        for (const kw of PRIORITY) {
          const buttons = root.querySelectorAll("button, a, [class*='btn']");
          for (const b of buttons) {
            if ((b.textContent || "").trim().includes(kw) &&
                !b.disabled && this._isElementClickable(b)) {
              return b;
            }
          }
        }
        return null;
      };

      return findBtn(card);
    }

    // 检查元素是否可见且可点击
    _isElementClickable(el) {
      if (!el) return false;
      if (el.disabled) return false;

      const style = window.getComputedStyle(el);
      if (style.display === "none" || style.visibility === "hidden" || style.opacity === "0") {
        return false;
      }

      const rect = el.getBoundingClientRect();
      // 检查元素是否在视口内或接近视口
      return !(rect.bottom < -50 || rect.top > window.innerHeight + 50 ||
               rect.right < -50 || rect.left > window.innerWidth + 50);
    }

  const AutomationEngine = {
    buildJobKey: (c, p) => {
      const idAttr = c?.dataset?.jobid || c?.dataset?.lid || c?.getAttribute("data-jobid") ||
                     c?.getAttribute("data-positionid") || c?.getAttribute("data-id") || "";
      // href 路径末段作为最终兜底，避免空 key 碰撞
      const hrefKey = c?.querySelector("a[href]")?.pathname?.split("/").filter(Boolean).pop() || "";
      const id = idAttr || hrefKey || `idx-${Array.from(c?.parentElement?.children || []).indexOf(c)}`;
      return `${id}|${p.title || ""}|${p.company || ""}`;

    refillJobQueue() {
      const s = state.settings;
      state.jobQueue = PlatformManager.adapter.getJobCards().filter(c => {
        const p = PlatformManager.adapter.parseJobCard(c), t = (p.title || "").toLowerCase(), l = (p.location || "").toLowerCase(), cName = (p.company || "").toLowerCase();
        const cardText = (c.textContent || "").toLowerCase();

        const apiData = APIDataCache.get(p.title + p.company) || APIDataCache.get(p.company);
        if (s.maxUpdateDays > 0 && apiData && apiData.timestamp) {
            const daysDiff = (Date.now() - apiData.timestamp) / (1000 * 60 * 60 * 24);
            if (daysDiff > s.maxUpdateDays) return false;
        }

        const btnText = ($("button, .btn, .op-btn-chat, .j_apply, a.op-btn-chat, a.btn-apply", c)?.textContent || "").trim();
        if (btnText.includes("继续沟通") || btnText.includes("已申请") || btnText.includes("已投递") || btnText.includes("已沟通") || btnText.includes("已回复")) {
            return false;
        }

        if (s.minSalaryK && s.minSalaryK > 0) {
            const salMatch = cardText.match(/(\d+)-(\d+)[k千w万]/i);
            if (salMatch) {
                let maxSal = parseInt(salMatch[2], 10);
                if(salMatch[0].match(/[w万]/i)) maxSal *= 10;
                if (maxSal < s.minSalaryK) return false;
            } else {
                const exactMatch = cardText.match(/(?:薪资|月薪)[:：]?\s*(\d+)[k千w万]/i);
                if (exactMatch) {
                    let exactSal = parseInt(exactMatch[1], 10);
                    if(exactMatch[0].match(/[w万]/i)) exactSal *= 10;
                    if(exactSal < s.minSalaryK) return false;
                }
            }
        }

        if (s.excludeDegreesStr) {
            const excludeDegs = Util.parseKw(s.excludeDegreesStr);
            if (excludeDegs.length && excludeDegs.some(deg => cardText.includes(deg))) return false;
        }

        if (s.includeKeywords.length && !s.includeKeywords.some(k => t.includes(k))) return false;
        if (s.excludeKeywords.length && s.excludeKeywords.some(k => t.includes(k) || cName.includes(k))) return false;
        if (s.locationKeywords.length && !s.locationKeywords.some(k => l.includes(k))) return false;
        if (s.excludeHeadhunters && p.isHeadhunter) return false;
        if (state.processedJobKeys.has(this.buildJobKey(c, p))) return false;
        return true;
      });
      state.currentIndex = 0;
      if (state.currentPlatform === APP.PLATFORM.ZHILIAN) {
        const sample = state.jobQueue.slice(0, 3).map(c => {
          const p = PlatformManager.adapter.parseJobCard(c);
          return `"${p.title||'(空)'}"→key:${this.buildJobKey(c,p)}`;
        });
        Logger.log(`[ZL] refillJobQueue: 队列=${state.jobQueue.length} 示例=[${sample.join(" | ")}]`);
      }
    },

          Logger.log(`[Auto] 循环异常: ${loopErr?.message || loopErr}`);

      const c = state.jobQueue[state.currentIndex];

      if (!document.body.contains(c)) {
          Logger.log(`[Auto] 卡片 #${state.currentIndex} 已从 DOM 脱离，清空队列重取`);

          state.currentIndex = 0;

          state.currentIndex++;

      Logger.log(`处理 ${state.currentIndex + 1}/${state.jobQueue.length}：${p.title}`);

          });

        } catch (err) {
          if (err.message === "LIMIT_REACHED") {
            Logger.log("平台提示已达今日投递上限，脚本自动停止。");
            Toast.show("今日投递已超过上限", "error");
            Toast.beep();
            this.stop();

          }

      } else {

          Toast.show(`连续${MAX_STREAK}张无按钮，已停止`, "error");
          this.stop();
          return;
        }
      }
      state.processedJobKeys.add(key); state.currentIndex++;
    },
    async processChat() {
      const mode = state.settings.communicationMode || "auto"; let cItem = null;
      if (mode === "auto") { const l = PlatformManager.adapter.getChatList(); if (l.length) { cItem = l[state.autoChatCursor % l.length]; state.autoChatCursor = (state.autoChatCursor + 1) % Math.max(l.length, 1); } }

      if (!cItem) { if (mode === "auto") { state.autoChatCursor = 0; await PlatformManager.adapter.scrollChatListForAutoMode(); } return; }
      const m = PlatformManager.adapter.getCurrentChatMeta(cItem), k = m.chatId ? `chat-${m.chatId}` : Util.toHrKey(m.hrName, m.companyName);
      if (!(await PlatformManager.adapter.openChat(cItem))) return;
      if (state.settings.communicationIncludeKeywords.length && !state.settings.communicationIncludeKeywords.some(kw => (PlatformManager.adapter.getPositionName() || "").toLowerCase().includes(kw))) { Logger.log("跳过：岗位不匹配"); if (mode === "auto") await PlatformManager.adapter.scrollChatListForAutoMode(); return; }

      const oSelf = PlatformManager.adapter.hasOnlySelfMessage();
      const pName = PlatformManager.adapter.getPositionName();

      if (oSelf && !state.interactions.sentGreet.has(k)) {
          let greeted = false;
          if (state.settings.customGreeting) {
              const processedGreet = Util.processTemplate(state.settings.customGreeting, m.hrName, pName, m.companyName);
              if (await PlatformManager.adapter.sendText(processedGreet)) {
                  greeted = true;
                  Logger.log("已发送自定义专属招呼");
              }
          } else {
              if (await PlatformManager.adapter.sendGreeting()) {
                  greeted = true;
                  Logger.log("已发官方招呼");
              }
          }
          if (greeted) {
              state.interactions.sentGreet.add(k);
              HistoryStore.flushSets();
          }
      }

      if (oSelf && state.settings.useAutoSendImageResume && await PlatformManager.adapter.sendImageResume()) { state.interactions.sentImage.add(k); HistoryStore.flushSets(); Logger.log("已发图简"); }
      if (PlatformManager.adapter.hasCardMessage() && await PlatformManager.adapter.acceptCardMessage()) Logger.log("已处理名片");

      if (oSelf && state.settings.useAutoSendTextResume && state.settings.resumeText && !state.interactions.sentTextResume.has(k)) {
          const processedText = Util.processTemplate(state.settings.resumeText, m.hrName, pName, m.companyName);
          if (await PlatformManager.adapter.sendText(processedText)) {
              state.interactions.sentTextResume.add(k);
              HistoryStore.flushSets();
              Logger.log("已发送文本简历(含魔法变量)");
          }
      }

      await Util.delay(APP.DELAY.MEDIUM, 200);
      const txt = Util.cleanMsg(await PlatformManager.adapter.getLastFriendMessageText());
      if (!txt || txt === state.lastProcessedMessage || state.processingMessage) return;
      state.processingMessage = true; state.lastProcessedMessage = txt;

      if (state.settings.useDesktopNotification && state.settings.notificationKeywords.length > 0) {
          const hitWord = state.settings.notificationKeywords.find(kw => txt.includes(kw));
          if (hitWord) {
              Toast.alertBeep();
              const notifText = `HR [${m.hrName}] 发来包含 "${hitWord}" 的消息：\n${txt}`;
              try {
                  GM_notification({ title: `Job Me - 收到重要消息`, text: notifText, timeout: 8000 });
              } catch(e) {
                  if (window.Notification && Notification.permission === "granted") {
                      new Notification("Job Me - 收到重要消息", { body: notifText });
                  }
              }
              Logger.log(`🔥 触发强提醒，捕获高价值消息 [${hitWord}]`);

          return;

      let replied = false;
      for (const r of state.settings.keywordReplies) {
        if (r.keyword && r.reply && txt.toLowerCase().includes(r.keyword.toLowerCase())) {
            const processedReply = Util.processTemplate(r.reply, m.hrName, pName, m.companyName);
            if (await PlatformManager.adapter.sendText(processedReply)) { Logger.log(`关键词回复已发送：${r.keyword}`); replied = true; break; }
        }
      }

      if (!replied && state.settings.useAiReply) {
          replied = await ConversationEngine.handleAiReply(txt);
      }

      if (!replied) {
          const mentionResume = (txt || "").includes("简历");
          if (mentionResume && state.settings.useAutoSendImageResume && !state.interactions.sentImage.has(k) && await PlatformManager.adapter.sendImageResume()) { state.interactions.sentImage.add(k); HistoryStore.flushSets(); Logger.log("已发送图片简历"); }
          else if (mentionResume && state.settings.useAutoSendResume && !state.interactions.sentResume.has(k) && await PlatformManager.adapter.sendResume()) { state.interactions.sentResume.add(k); HistoryStore.flushSets(); Logger.log("已发送附件简历"); }
          else if (mentionResume && state.settings.useAutoSendTextResume && state.settings.resumeText && !state.interactions.sentTextResume.has(k)) {
              const processedText = Util.processTemplate(state.settings.resumeText, m.hrName, pName, m.companyName);
              if (await PlatformManager.adapter.sendText(processedText)) {
                  state.interactions.sentTextResume.add(k); HistoryStore.flushSets(); Logger.log("已发送文本简历(含魔法变量)");
              }
          }
      }

      state.processingMessage = false; state.interactions.processedHR.add(k); HistoryStore.flushSets(); Logger.log(`处理完成：${m.hrName}`);
      if (mode === "auto") await PlatformManager.adapter.scrollChatListForAutoMode();
    }

  function bossApplyFilters() {
    if (!GM_getValue("boss_af_enabled", false)) return;
    const toClick = [];
    const debugSteps = [];

    const check = (cat, val) => {
      if (val === "不限") return;
      const ka = BOSS_FILTER_KA[cat]?.[val];
      if (!ka) { debugSteps.push(`${cat}: 无映射值 "${val}"`); return; }

      // 主选择器：ka 属性精确匹配
      let li = document.querySelector(`[ka="sel-job-rec-${ka}"]`);

      // 备选：在筛选区域用文本内容匹配
      if (!li) {
        const filterArea = document.querySelector(
          '[class*="search-condition"], [class*="filter-bar"], [class*="filter-wrap"], [class*="job-search-filter"]'
        );
        if (filterArea) {
          li = Array.from(filterArea.querySelectorAll('li, a, span, button, label'))
            .find(el => el.textContent.trim() === val && !el.children.length);
        }
      }

      // 最终备选：全文档文本匹配（限定小节点）
      if (!li) {
        li = Array.from(document.querySelectorAll('[class*="filter"] li, [class*="filter"] a, [class*="filter"] span'))
          .find(el => el.textContent.trim() === val && !el.querySelector('li, a'));
      }

      if (li) {
        if (!li.classList.contains("active")) toClick.push(li);
        debugSteps.push(`${cat} "${val}": ✓ <${li.tagName}> ${(li.className||'').slice(0,30)}`);
      } else {
        debugSteps.push(`${cat} "${val}": ✗ 未找到（ka="${ka}"）`);

  // 智联招聘增强：自动应用筛选条件
  // 策略：全页文本扫描 trigger → click → 等待 option → click → CSS 注入兜底
  // ------------------------------------------------------------------
  async function zhilianApplyFilters() {
    if (!GM_getValue("zl_af_enabled", false)) return;
    Logger.log("[ZL-Filter] 开始应用筛选...");
    let appliedCount = 0;

    const FILTER_DEF = [
      { labels: ["薪资",    "薪资待遇", "月薪",    "薪资范围"],         value: GM_getValue("zl_af_salary",  "不限") },
      { labels: ["学历",    "学历要求", "最低学历"],                     value: GM_getValue("zl_af_degree",  "不限") },
      { labels: ["经验",    "工作经验", "经验要求"],                     value: GM_getValue("zl_af_exp",     "不限") },
      { labels: ["职位类型","工作类型", "类型",   "工作性质"],           value: GM_getValue("zl_af_jobtype", "不限") },
      { labels: ["公司性质","企业性质", "性质",   "单位性质"],           value: GM_getValue("zl_af_nature",  "不限") },
      { labels: ["公司规模","规模",     "人员规模","企业规模"],           value: GM_getValue("zl_af_scale",   "不限") },
      { labels: ["融资阶段","融资",     "融资情况"],                     value: GM_getValue("zl_af_funding", "不限") },
    ];

    // ── 查 trigger：全页面不限容器，按文本精确/前缀匹配 ──
    // 先搜 li/a/span/button/label/em，再兜底搜单叶 div
    const findTrigger = (labels) => {
      const inline = Array.from(document.querySelectorAll(
        'li, a, span, button, label, em, [role="button"], [role="tab"], [role="menuitem"]'
      ));
      for (const label of labels) {
        const el = inline.find(e => {
          const t = (e.textContent || "").trim();
          return t === label || (t.startsWith(label) && t.length <= label.length + 8);
        });
        if (el) return el;
      }
      // 兜底：只有 0-1 个子元素的 div
      for (const label of labels) {
        const el = Array.from(document.querySelectorAll("div")).find(e => {
          if (e.children.length > 1) return false;
          const t = (e.textContent || "").trim();
          return t === label || (t.startsWith(label) && t.length <= label.length + 8);
        });
        if (el) return el;
      }
      return null;
    };

    // ── 查 option：trigger click 后全页面精确匹配目标值 ──
    const findOption = (value) => {
      const el = Array.from(document.querySelectorAll(
        'li, a, span, button, label, em, p, [role="option"], [role="menuitem"]'
      )).find(e => (e.textContent || "").trim() === value);
      if (el) return el;
      // 兜底单叶 div
      return Array.from(document.querySelectorAll("div")).find(e => {
        return e.children.length <= 1 && (e.textContent || "").trim() === value;
      }) || null;
    };

    // ── CSS 注入：强制展开所有可能隐藏的下拉面板 ──
    const STYLE_ID = "jm-zl-fo";
    const injectCSS = () => {
      if (document.getElementById(STYLE_ID)) return;
      const s = document.createElement("style");
      s.id = STYLE_ID;
      s.textContent = `
        [class*="filter"] *, [class*="Filter"] *,
        [class*="condition"] *, [class*="Condition"] *,
        [class*="sift"] *, [class*="refine"] *,
        [class*="dropdown"] *, [class*="Dropdown"] *,
        [class*="panel"] *, [class*="Panel"] *,
        [class*="select-list"] *, [class*="option-list"] *
        { display: revert !important; visibility: visible !important;
          opacity: 1 !important; height: auto !important;
          max-height: none !important; overflow: visible !important;
          pointer-events: auto !important; }
      `;
      document.head.appendChild(s);
    };
    const removeCSS = () => { const s = document.getElementById(STYLE_ID); if (s) s.remove(); };

    for (const f of FILTER_DEF) {
      if (!f.value || f.value === "不限") continue;
      Logger.log(`[ZL-Filter] 处理: ${f.labels[0]} → "${f.value}"`);
      try {
        // Step 1: 找 trigger → click → 等待 option 进入 DOM（最多 2s）
        const trigger = findTrigger(f.labels);
        if (trigger) {
          Logger.log(`[ZL-Filter] trigger <${trigger.tagName}> "${(trigger.textContent||"").trim().slice(0,12)}"`);
          trigger.click();
          for (let i = 0; i < 10; i++) {
            await Util.delay(200);
            if (findOption(f.value)) break;
          }
        } else {
          Logger.log(`[ZL-Filter] trigger 未找到: ${f.labels.join("/")}`);
        }

        // Step 2: 找 option
        let option = findOption(f.value);

        // Step 3: CSS 注入兜底（option 在 DOM 但被 display:none 隐藏）
        if (!option) {
          injectCSS();
          await Util.delay(800);
          option = findOption(f.value);
        }

        if (option) {
          option.click();
          appliedCount++;
          Logger.log(`[ZL-Filter] ✓ 已选: "${f.value}" <${option.tagName}>`);
          removeCSS();
          await Util.delay(500);
          // 等 SPA 刷新后筛选栏重新出现（最多 5s）
          for (let w = 0; w < 25; w++) {
            if (document.querySelector('[class*="filter"],[class*="Filter"],[class*="condition"],[class*="screen"]')) break;
            await Util.delay(200);
          }
        } else {
          Logger.log(`[ZL-Filter] ✗ option 未找到: "${f.value}"`);
          removeCSS();
        }
      } catch (err) {
        removeCSS();
        Logger.log(`[ZL-Filter] ✗ 异常 "${f.labels[0]}": ${err.message}`);
      }
    }

    removeCSS();
    Logger.log(`[ZL-Filter] 完成，共应用 ${appliedCount} 项筛选`);
    _zlFilterRunning = false;

    // 有设置但未全部应用 → 弹出诊断面板
    const hasSettings = FILTER_DEF.some(f => f.value && f.value !== "不限");
    const wantCount   = FILTER_DEF.filter(f => f.value && f.value !== "不限").length;
    if (hasSettings && appliedCount < wantCount) {
      _showZlFilterDebugPanel(FILTER_DEF);
    }

    // 筛选完成后自动启动投递（需开启"自动投递"且未在运行中）
    if (appliedCount > 0 && GM_getValue("zl_af_autostart", true) && !state.isRunning) {
      Logger.log("[ZL-Filter] 筛选完毕，等待页面稳定后自动启动投递...");
      await Util.delay(2000);
      if (!state.isRunning) {
        Logger.log("[ZL-Filter] 自动启动 AutomationEngine...");
        AutomationEngine.start();
      }
    }
  }

    const enabled = GM_getValue("zl_af_enabled", false);
    Logger.log("[ZL-Filter] page=" + location.pathname + " enabled=" + enabled + " running=" + _zlFilterRunning);

    setTimeout(() => {
      const p = location.pathname.toLowerCase();
      const isRecommend = p === "/recommend" || p.startsWith("/recommend");
      const hasSou = location.host.includes("sou.zhaopin.com");
      if (isRecommend && !hasSou) {
        // 检查是否有筛选栏；没有则重定向到搜索页并带参数
        const WAIT_SEL = [
          '[class*="filter"]','[class*="Filter"]','[class*="condition"]',
          '[class*="screen"]','[class*="Screen"]','[class*="sift"]',
          '[class*="tag-bar"]','[class*="tagBar"]',
        ].join(',');
        const hasBar = !!document.querySelector(WAIT_SEL);
        Logger.log("[ZL-Filter] /recommend 筛选栏存在: " + hasBar);
        if (!hasBar) {
          // 构造带筛选参数的搜索 URL 并跳转
          const kw = GM_getValue("boss_tab_name", "") || GM_getValue("zl_kw", "");
          const salary = GM_getValue("zl_af_salary", "不限");
          const exp    = GM_getValue("zl_af_exp",    "不限");
          const degree = GM_getValue("zl_af_degree", "不限");
          // 构建 sou.zhaopin.com 参数（salary/exp/degree 用中文值，zhaopin 搜索支持）
          const params = new URLSearchParams();
          if (kw) params.set("kw", kw);
          if (salary && salary !== "不限") params.set("sl", salary);
          if (exp    && exp    !== "不限") params.set("we", exp);
          if (degree && degree !== "不限") params.set("el", degree);
          const target = "https://sou.zhaopin.com/?" + params.toString();
          Logger.log("[ZL-Filter] /recommend 无筛选栏，跳转到搜索页: " + target);
          location.href = target;
          _zlFilterRunning = false;
          return;
        }
      }
      zhilianTryApplyFilters();
      setTimeout(() => { _zlFilterRunning = false; }, 30000);
    }, 1200);
  }

  }
